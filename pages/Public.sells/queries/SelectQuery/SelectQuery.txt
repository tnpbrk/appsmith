SELECT sell_id, sell_date, customer_name, SUM(sell_qty * shoe_price) AS total
FROM sells
JOIN customers USING (customer_id)
LEFT JOIN sell_details USING (sell_id)
LEFT JOIN shoes USING (shoe_id)
GROUP BY sell_id, sell_date, customer_name
ORDER BY {{data_table.sortOrder.column || 'sell_id'}} {{data_table.sortOrder.order || "ASC"}}
LIMIT {{data_table.pageSize}}
OFFSET {{(data_table.pageNo - 1) * data_table.pageSize}};
